% 5. Answer fallback: existing safeguard and its limitation
\begin{frame}[fragile]{05 \quad Answer fallback: activation is not submission}
\small\textcolor{muted}{A safeguard after Deriver can activate Answer, but cannot force a final submission.}
\begin{columns}[T,totalwidth=\textwidth]
\column{0.49\textwidth}
\textbf{Source code (abbreviated)}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{7.2}{8.5}\selectfont]
_WAKE_AFTER = {"deriver", "verifier"}
_FALLBACK_AFTER = {"deriver"}

if last_role not in _WAKE_AFTER:
    return False

llm_decision = super().match_wakeup_condition(env)
if llm_decision:
    return True
if last_role in _FALLBACK_AFTER:
    return True  # LLM said NO
return False
\end{lstlisting}
\column{0.49\textwidth}
\textbf{Potential failure sequence}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{7.2}{8.5}\selectfont]
Deriver contributes
       v
Answer NO -> fallback YES
       v
Answer emits <step> only
       v
Next turn: Answer blocked
Other agents also say NO
       v
NO_ACTIVE_AGENTS
\end{lstlisting}
\vspace{2mm}
\textcolor{muted}{When context is insufficient, Answer may provide a non-final step instead.}
\end{columns}
\vspace{-1mm}
\observation{Fallback may postpone, rather than eliminate, a deadlock: activation does not guarantee submission.}
\source{EoM researchworld/agent.py and prompts.py. E1 continuation shows Answer $\to$ no-active, but does not prove fallback woke Answer.}
\end{frame}

% 6. Fixed-answer prompt: shared instructions
\begin{frame}[fragile]{07 \quad Fixed Answer: Prompt Template}
\small\textcolor{muted}{The same evaluator prompt is used for Empty, Base, and Treated states (12,000-token output cap).}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{8.0}{9.6}\selectfont]
You are producing a final answer to a multi-part scientific reasoning problem.

{state}

INSTRUCTIONS:
- Answer each sub-question in order. Include essential reasoning, equations, and numerical results.
- Do not restate the problem or add an introductory recap.
- Be concise: avoid repeating prior explanations or adding a final summary.
- Shared notes may contain errors; check what you use.
{mode_rule}
- For parts you cannot solve, identify the missing result concisely. Do not refuse the task.
- Output plain text (no XML tags).
- After the LAST sub-question, end with a separate final line exactly:
END_OF_SOLUTION
\end{lstlisting}
\vspace{2mm}\small\textcolor{muted}{\texttt{\{state\}}: problem + shared notes; \texttt{\{mode\_rule\}}: Full or Notes-only instruction.}
\source{Fixed-answer evaluation prompt, used in the counterfactual scoring scripts.}
\end{frame}

% 7. Fixed-answer prompt: mode-dependent instructions
\begin{frame}[fragile]{08 \quad Fixed Answer: Full vs. Notes-Only}
\small\textcolor{muted}{Exactly one of these instructions replaces \texttt{\{mode\_rule\}} in the shared prompt.}
\par\vspace{3mm}
\textbf{Full mode}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{8.3}{10.4}\selectfont]
- You MAY derive missing steps yourself; mark uncertain conclusions [UNCERTAIN].
\end{lstlisting}
\vspace{6mm}
\textbf{Notes-only mode}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{8.3}{10.4}\selectfont]
- Use ONLY facts and results explicitly present in shared notes. Do NOT add new derivations. For each missing part write [NOT COVERED BY NOTES].
\end{lstlisting}
\vspace{7mm}
\observation{Both modes use the same model and common prompt. The notes-only constraint is a prompting instruction, not a guaranteed restriction on the model's reasoning.}
\source{Fixed-answer evaluation prompt, mode-specific instructions.}
\end{frame}
